function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // Busca flexível de abas mesmo se houver diferença de maiúsculas/minúsculas
    function findSheet(name) {
      var sheets = ss.getSheets();
      for (var i = 0; i < sheets.length; i++) {
        if (sheets[i].getName().trim().toUpperCase() === name.toUpperCase()) {
          return sheets[i];
        }
      }
      return null;
    }

    var clientsSheet = findSheet("CLIENTES");
    var ordersSheet = findSheet("PEDIDOS");
    var paymentsSheet = findSheet("PAGAMENTOS");

    var db = {
      status: "success",
      timestamp: new Date().toISOString(),
      clients: getSheetData(clientsSheet),
      orders: getSheetData(ordersSheet),
      payments: getSheetData(paymentsSheet)
    };

    return ContentService.createTextOutput(JSON.stringify(db))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
    .setMimeType(ContentService.MimeType.JSON);
}

function getSheetData(sheet) {
  if (!sheet) return [];
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];

  // Localiza linha com cabeçalhos reais
  var headerRowIdx = 0;
  for (var r = 0; r < Math.min(5, data.length); r++) {
    var nonEmpty = data[r].filter(function(cell) { return cell !== "" && cell !== null; });
    if (nonEmpty.length >= 2) {
      headerRowIdx = r;
      break;
    }
  }

  var headers = data[headerRowIdx];
  var rows = [];

  for (var i = headerRowIdx + 1; i < data.length; i++) {
    var row = data[i];
    var hasContent = row.some(function(cell) { return cell !== "" && cell !== null; });
    if (!hasContent) continue;

    var obj = {};
    for (var h = 0; h < headers.length; h++) {
      var headerName = String(headers[h]).trim();
      if (!headerName) continue;
      
      var val = row[h];
      // Tratamento de datas para formato limpo YYYY-MM-DD
      if (val instanceof Date) {
        var y = val.getFullYear();
        var m = ("0" + (val.getMonth() + 1)).slice(-2);
        var d = ("0" + val.getDate()).slice(-2);
        val = y + "-" + m + "-" + d;
      }
      obj[headerName] = val;
    }
    rows.push(obj);
  }
  return rows;
}
